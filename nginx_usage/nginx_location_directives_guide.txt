# Nginx Location 块中每个指令的详细说明

server {
    listen 80;
    server_name     example.com;
    root /var/www/html;  ###  设置文档根目录
    
    # ==================== Location 匹配语法 ====================
    
    # 精确匹配 (=)
    location = / {
        # 只匹配根路径 "/"，优先级最高
        return 200 "Exact match for root";
    }
    
    # 优先前缀匹配 (^~)
    location ^~ /images/ {
        # 匹配以 /images/ 开头的路径，不再检查正则表达式
        root /var/www;
        # 匹配: /images/logo.png, /images/photos/1.jpg
        # 不匹配: /img/logo.png
    }
    
    # 正则匹配 - 区分大小写 (~)
    location ~ \.(gif|jpg|png)$ {
        # 匹配以 .gif/.jpg/.png 结尾的文件
        root /var/www/images;
        # 匹配: /photo.jpg, /logo.PNG (不匹配，区分大小写)
    }
    
    # 正则匹配 - 不区分大小写 (~*)
    location ~* \.(gif|jpg|png|jpeg)$ {
        # 匹配图片文件，不区分大小写
        root /var/www/images;
        expires 30d;
        # 匹配: /photo.JPG, /logo.png, /image.GIF
    }
    
    # 前缀匹配（默认）
    location /api/ {
        # 匹配以 /api/ 开头的所有路径
        proxy_pass http://backend;
        # 匹配: /api/users, /api/orders, /api/v1/products
    }
    
    # 命名location (@)
    location @fallback {
        # 只能用于内部重定向
        proxy_pass http://backend;
    }
    
    # ==================== 文件系统相关指令 ====================
    
    location /static/ {
        # root - 设置根目录
        root /var/www;                          # 实际路径: /var/www/static/
        
        # alias - 设置别名（替换匹配部分）
        # alias /var/www/assets/;               # 实际路径: /var/www/assets/
        
        # index - 默认索引文件
        index index.html index.htm default.html;
        
        # autoindex - 目录列表
        autoindex on;                           # 启用目录浏览
        autoindex_exact_size off;               # 显示文件大小的格式
        autoindex_localtime on;                 # 使用本地时间
        
        # try_files - 文件查找顺序
        try_files $uri $uri/ /index.html;      # 依次尝试文件、目录、最后fallback
        try_files $uri $uri/ @fallback;        # 最后重定向到命名location
        try_files $uri =404;                   # 找不到返回404
        
        # open_file_cache - 文件缓存
        open_file_cache max=1000 inactive=20s;  # 缓存1000个文件，20秒未访问删除
        open_file_cache_valid 30s;              # 缓存有效期30秒
        open_file_cache_min_uses 2;             # 至少访问2次才缓存
        open_file_cache_errors on;              # 缓存文件错误信息
    }
    
    # ==================== 访问控制指令 ====================
    
    location /admin/ {
        # allow/deny - IP访问控制
        allow 192.168.1.0/24;                  # 允许内网访问
        allow 127.0.0.1;                       # 允许本机访问
        deny all;                              # 拒绝其他所有IP
        
        # satisfy - 访问控制组合模式
        satisfy any;                           # any: IP或HTTP认证任一通过即可
        # satisfy all;                         # all: IP和HTTP认证都必须通过
        
        # HTTP基础认证
        auth_basic "Restricted Area";          # 认证提示信息
        auth_basic_user_file /etc/nginx/.htpasswd; # 用户密码文件
        
        proxy_pass http://admin_backend;
    }
    
    # ==================== 请求限制指令 ====================
    
    location /upload/ {
        # client_max_body_size - 请求体大小限制
        client_max_body_size 100M;             # 允许上传100MB文件
        
        # client_body_timeout - 请求体超时
        client_body_timeout 300s;              # 5分钟内必须上传完成
        
        # client_body_temp_path - 临时文件路径
        client_body_temp_path /var/nginx/tmp;  # 临时文件存储位置
        
        # client_body_in_file_only - 强制将请求体写入文件
        client_body_in_file_only clean;        # clean: 请求结束后删除临时文件
        
        proxy_pass http://upload_backend;
    }
    
    # ==================== 响应处理指令 ====================
    
    location /content/ {
        # add_header - 添加响应头
        add_header X-Content-Source "nginx" always;
        add_header Cache-Control "public, max-age=3600";
        add_header X-Frame-Options SAMEORIGIN;
        
        # more_set_headers - 更灵活的头部设置（需要 headers-more-nginx-module）
        # more_set_headers "Server: MyServer";
        
        # expires - 缓存过期时间
        expires 1h;                            # 1小时后过期
        expires modified +1h;                  # 基于文件修改时间+1小时
        expires @23h59m;                       # 每天23:59过期
        expires -1;                            # 立即过期
        expires epoch;                         # 设置为1970-01-01
        expires off;                           # 不设置过期头
        
        root /var/www;
    }
    
    # ==================== URL重写和重定向 ====================
    
    location /old-section/ {
        # return - 返回状态码或重定向
        return 301 /new-section/;              # 永久重定向
        return 302 /temporary-moved/;           # 临时重定向
        return 404;                            # 返回404状态
        return 200 "Custom content";           # 返回自定义内容
        return 200 "application/json" '{"status": "ok"}'; # 带Content-Type
    }
    
    location /products/ {
        # rewrite - URL重写
        rewrite ^/products/(\d+)$ /product.php?id=$1 last;    # 重写后继续匹配location
        rewrite ^/products/(\w+)$ /category.php?name=$1 break; # 重写后不再匹配
        rewrite ^/old-products/(.*)$ /products/$1 permanent;   # 永久重定向
        rewrite ^/temp-products/(.*)$ /products/$1 redirect;   # 临时重定向
        
        root /var/www;
    }
    
    # ==================== 条件判断和变量 ====================
    
    location /mobile/ {
        # set - 设置变量
        set $mobile 0;
        set $mobile_rewrite do_not_perform;
        
        # if - 条件判断（谨慎使用）
        if ($http_user_agent ~* "(android|bb\d+|meego).+mobile") {
            set $mobile 1;
            set $mobile_rewrite perform;
        }
        
        if ($mobile_rewrite = perform) {
            rewrite ^(.*)$ /mobile$1 last;
        }
        
        if ($request_method = POST) {
            return 405;
        }
        
        if ($args ~ "debug=1") {
            access_log /var/log/nginx/debug.log;
        }
        
        root /var/www;
    }
    
    # ==================== 代理配置指令 ====================
    
    location /api/ {
        # proxy_pass - 代理后端服务
        proxy_pass http://backend;              # 代理到upstream
        proxy_pass http://127.0.0.1:3000;      # 代理到具体地址
        proxy_pass http://backend/app/;         # 替换路径前缀
        
        # 请求头设置
        proxy_set_header Host $host;                    # 传递原始Host头
        proxy_set_header X-Real-IP $remote_addr;       # 客户端真实IP
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;    # 协议类型
        proxy_set_header X-Forwarded-Host $host;       # 原始主机名
        proxy_set_header X-Forwarded-Port $server_port; # 原始端口
        
        # 连接和超时设置
        proxy_connect_timeout 30s;             # 连接后端超时
        proxy_send_timeout 30s;                # 发送请求超时
        proxy_read_timeout 30s;                # 读取响应超时
        proxy_next_upstream_timeout 10s;       # 故障转移总超时时间
        
        # 缓冲设置
        proxy_buffering on;                     # 启用响应缓冲
        proxy_buffer_size 4k;                  # 响应头缓冲区大小
        proxy_buffers 8 4k;                    # 响应体缓冲区数量和大小
        proxy_busy_buffers_size 8k;            # 忙碌缓冲区大小
        
        # 请求体处理
        proxy_request_buffering on;            # 启用请求缓冲
        proxy_max_temp_file_size 1024m;        # 临时文件最大大小
        
        # 故障转移
        proxy_next_upstream error timeout invalid_header http_500 http_502 http_503 http_504;
        proxy_next_upstream_tries 3;           # 最多重试3次
        
        # HTTP版本和连接
        proxy_http_version 1.1;                # 使用HTTP/1.1
        proxy_set_header Connection "";        # 启用连接池
        
        # WebSocket支持
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        
        # SSL代理（当后端使用HTTPS时）
        # proxy_ssl_verify off;                # 不验证后端SSL证书
        # proxy_ssl_session_reuse on;          # 重用SSL会话
    }
    
    # ==================== FastCGI配置（PHP）====================
    
    location ~ \.php$ {
        # FastCGI基本配置
        fastcgi_pass unix:/var/run/php-fpm.sock;    # PHP-FPM socket
        fastcgi_pass 127.0.0.1:9000;                # PHP-FPM TCP
        
        # FastCGI参数
        fastcgi_index index.php;                     # 默认脚本
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        include fastcgi_params;                      # 包含标准参数
        
        # FastCGI超时设置
        fastcgi_connect_timeout 60s;
        fastcgi_send_timeout 60s;
        fastcgi_read_timeout 60s;
        
        # FastCGI缓冲
        fastcgi_buffering on;
        fastcgi_buffer_size 4k;
        fastcgi_buffers 8 4k;
        
        # FastCGI缓存（可选）
        # fastcgi_cache WORDPRESS;
        # fastcgi_cache_valid 200 60m;
    }
    
    # ==================== 静态文件优化 ====================
    
    location ~* \.(css|js|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        # 文件位置
        root /var/www/assets;
        
        # 缓存设置
        expires 1y;                             # 1年过期
        add_header Cache-Control "public, immutable";
        
        # 性能优化
        access_log off;                         # 关闭访问日志
        log_not_found off;                      # 不记录404错误
        
        # 压缩（通常在http块设置）
        gzip_static on;                         # 如果存在.gz文件直接返回
        
        # 跨域设置
        add_header Access-Control-Allow-Origin *;
        
        # 防盗链
        valid_referers none blocked server_names *.example.com example.*;
        if ($invalid_referer) {
            return 403;
        }
    }
    
    # ==================== 安全相关配置 ====================
    
    # 禁止访问隐藏文件
    location ~ /\. {
        deny all;
        access_log off;
        log_not_found off;
    }
    
    # 禁止访问备份文件
    location ~ ~$ {
        deny all;
        access_log off;
        log_not_found off;
    }
    
    # 禁止访问敏感文件
    location ~* \.(conf|sql|bak|old)$ {
        deny all;
    }
    
    # 限制特定路径的请求方法
    location /api/admin/ {
        limit_except GET POST {
            deny all;
        }
        proxy_pass http://admin_backend;
    }
    
    # ==================== 日志配置 ====================
    
    location /logs/ {
        # access_log - 访问日志
        access_log /var/log/nginx/special.log combined;
        access_log /var/log/nginx/special.log main buffer=64k flush=5s;
        access_log syslog:server=192.168.1.1 combined;
        access_log off;                         # 关闭日志
        
        # error_log - 错误日志（继承server设置）
        error_log /var/log/nginx/location_error.log warn;
        
        # log_not_found - 404错误记录
        log_not_found off;                      # 不记录404到error_log
        
        return 403;
    }
    
    # ==================== 限流和并发控制 ====================
    
    location /api/sensitive/ {
        # limit_req - 请求限流（需要在http块定义limit_req_zone）
        limit_req zone=api burst=10 nodelay;   # 使用api限流区域
        
        # limit_conn - 连接限制（需要在http块定义limit_conn_zone）
        limit_conn perip 5;                    # 每个IP最多5个连接
        
        proxy_pass http://backend;
    }
    
    # ==================== 内容处理 ====================
    
    location /transform/ {
        # sub_filter - 内容替换（需要 sub_module）
        sub_filter 'old_text' 'new_text';
        sub_filter_once off;                    # 替换所有匹配项
        
        # charset - 字符集覆盖
        charset utf-8;
        override_charset on;
        
        proxy_pass http://backend;
    }
    
    # ==================== 错误处理 ====================
    
    location /special/ {
        # error_page - location级别的错误页面
        error_page 404 /custom_404.html;
        error_page 500 502 503 504 @error;
        
        # proxy_intercept_errors - 拦截后端错误
        proxy_intercept_errors on;             # 使用nginx错误页面而不是后端错误页面
        
        proxy_pass http://backend;
    }
    
    # ==================== 完整的API代理示例 ====================
    
    location /api/v1/ {
        # 访问控制
        allow 192.168.1.0/24;
        deny all;
        
        # 请求限制
        client_max_body_size 10M;
        client_body_timeout 30s;
        
        # 代理配置
        proxy_pass http://api_backend/;
        proxy_http_version 1.1;
        
        # 请求头
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header Connection "";
        
        # 超时设置
        proxy_connect_timeout 10s;
        proxy_send_timeout 10s;
        proxy_read_timeout 30s;
        
        # 缓冲设置
        proxy_buffering on;
        proxy_buffer_size 4k;
        proxy_buffers 8 4k;
        
        # 故障转移
        proxy_next_upstream error timeout invalid_header http_500 http_502 http_503;
        proxy_next_upstream_tries 2;
        
        # 响应头
        add_header X-Cache-Status $upstream_cache_status;
        add_header X-Response-Time $request_time;
        
        # 错误处理
        error_page 502 503 504 @api_error;
        proxy_intercept_errors on;
        
        # 日志
        access_log /var/log/nginx/api.log combined;
    }
    
    # API错误处理
    location @api_error {
        add_header Content-Type application/json;
        return 503 '{"error": "Service temporarily unavailable"}';
    }
}
